\cvsubsection{Formal Employment Projects}

    \begin{cvtable}[3]

        \cvitem{2022 - currently}{\href{https://3domics.eu}{3D'omics}}{Post-doc}
        {Aim of the project is to come up with technology and protocols to generate 3D omics landscapes,
        achieving reconstructions of intestinal host microbiota ecosystems.
        My role on the project is the analysis of microbial networks integrating spatial and metabolic data 
        to predict species interactions and/or influence of metabolites on species distributions
        using co-occurrence network analysis and Genome-Scale Metabolic Modeling.
        }

        \cvitem{2022 - 2022}{\href{https://imbbc.hcmr.gr/project/eosc-life-gos/}{
            A workflow for marine Genomic Observatories data analysis}}{Research associate}
            {An \href{https://www.eosc-life.eu}{EOSC-Life}-supported project aimed at making the large volumes of 
            data produced by genomic observatories more easily interpretable by providing the taxonomic inventories 
            of each sample in a timely and non-technical format. I was the lead developer of the delivered workflow, 
            and the Principal Investigator in the second half of the project.
        }

        \cvitem{2018 - 2021}{\href{http://prego.hcmr.gr/}{PREGO: Process, environment, organism (PREGO)}}{PhD}
            {PREGO is a systems-biology approach to elucidate ecosystem function at the microbial dimension.
            PREGO was the main project of my PhD and my role was to develop the data integration parts of the platform.
        }
        
        \cvitem{2019 -- present}{\href{https://elixir-greece.org}{ELIXIR–GR}}{technical support}
            {ELIXIR-GR is the Greek National Node of the ESFRI \href{https://elixir-europe.org/}{European RI ELIXIR}, 
            a distributed e-Infrastructure aiming at the construction of a sustainable European infrastructure for biological information.
            My role on the project was to develop containerized-versions of various software tools.
        }
        
        \cvitem{2018 - 2020}{\href{https://reconnect.hcmr.gr/}{RECONNECT}}{PhD}
            {RECONNECT is an Interreg V-B "Balkan-Mediterranean 2014-2020" project. It aims to develop strategies 
            for sustainable management of Marine Protected Areas (MPAs) and Natura 2000 sites.
        }

    \end{cvtable}
